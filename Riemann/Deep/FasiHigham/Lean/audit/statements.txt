-- Defs.lean
/-- Coefficient of z^j in q_{k,m}: (−1)^j C(m,j) (k+m−j)!/(k+m)! for j ≤ m, and 0 for j > m. -/
def qcoef (k m j : ℕ) : ℚ :=
  if j ≤ m then (-1) ^ j * (m.choose j : ℚ) * ((k + m - j).factorial : ℚ) / ((k + m).factorial : ℚ)
  else 0

-- Defs.lean
/-- Shifted coefficients: cs k 0 = c_{−2} = 0, cs k 1 = c_{−1} = 0, cs k 2 = c_0 = 1 and, for n ≥ 0,
c_{n+1} = 3/(k+3) c_n − 3/((k+2)(k+3)) c_{n−1} + 1/((k+1)(k+2)(k+3)) c_{n−2}. -/
def cs (k : ℕ) : ℕ → ℚ
  | 0 => 0
  | 1 => 0
  | 2 => 1
  | n + 3 => 3 / (k + 3) * cs k (n + 2) - 3 / ((k + 2) * (k + 3)) * cs k (n + 1)
      + 1 / ((k + 1) * (k + 2) * (k + 3)) * cs k n

/-- c_n(k), the n-th Taylor coefficient of 1/q_{k,3} at 0 (see conv and PS.lean). -/
def c (k n : ℕ) : ℚ := cs k (n + 2)

-- Defs.lean
/-- c_n(k), the n-th Taylor coefficient of 1/q_{k,3} at 0 (see conv and PS.lean). -/
def c (k n : ℕ) : ℚ := cs k (n + 2)

-- Defs.lean
/-- q_{k,3} · Σ_n c_n z^n = 1, coefficientwise: Σ_{j ≤ n} q_j c_{n−j} = 1 if n = 0, else 0. -/
theorem conv (k n : ℕ) :
    ∑ j ∈ range (n + 1), qcoef k 3 j * c k (n - j) = if n = 0 then 1 else 0 := by

-- PS.lean
/-- q_{k,m} as a formal power series over ℚ (a polynomial: coefficients vanish beyond z^m). -/
noncomputable def qPS (k m : ℕ) : PowerSeries ℚ := PowerSeries.mk (qcoef k m)

-- PS.lean
/-- c_n(k) = [z^n] (q_{k,3})⁻¹ in PowerSeries ℚ. -/
theorem coeff_inv (k n : ℕ) : PowerSeries.coeff n (qPS k 3)⁻¹ = c k n := by

-- PS.lean
/-- Uniqueness: every b with Σ_{j ≤ n} q_j b_{n−j} = [n = 0] for all n is c. -/
theorem conv_unique (k : ℕ) (b : ℕ → ℚ)
    (hb : ∀ n, ∑ j ∈ range (n + 1), qcoef k 3 j * b (n - j) = if n = 0 then 1 else 0) :

-- Main.lean
/-- With N = k + 1 and the root d: v = |z_±|², u = Re z_±, R₀ = v/(3(d² + N)) (the residue at z₀ = N + d),
K = R₀² · 3z₀²(d² + N)/(v(v − u²)) (= (2|R₊|)², Quant.lean). -/
noncomputable def vR (N d : ℝ) : ℝ := N ^ 2 - d * N + d ^ 2 + 3 * N
noncomputable def uR (N d : ℝ) : ℝ := N - d / 2
noncomputable def R0 (N d : ℝ) : ℝ := vR N d / (3 * (d ^ 2 + N))
noncomputable def KR (N d : ℝ) : ℝ :=
  R0 N d ^ 2 * (3 * (N + d) ^ 2 * (d ^ 2 + N)) / (vR N d * (vR N d - uR N d ^ 2))

-- Main.lean
noncomputable def uR (N d : ℝ) : ℝ := N - d / 2
noncomputable def R0 (N d : ℝ) : ℝ := vR N d / (3 * (d ^ 2 + N))
noncomputable def KR (N d : ℝ) : ℝ :=
  R0 N d ^ 2 * (3 * (N + d) ^ 2 * (d ^ 2 + N)) / (vR N d * (vR N d - uR N d ^ 2))

-- Quant.lean
/-- q_{k,3}(z) for real z, from the general coefficient formula. -/
noncomputable def qR (k : ℕ) (z : ℝ) : ℝ := ∑ j ∈ range 4, (qcoef k 3 j : ℝ) * z ^ j

-- Quant.lean
/-- The derivative q_{k,3}'(z) (termwise). -/
noncomputable def qR' (k : ℕ) (z : ℝ) : ℝ := ∑ j ∈ range 4, (j : ℝ) * (qcoef k 3 j : ℝ) * z ^ (j - 1)

-- Quant2.lean
/-- QUANTITATIVE THEOREM. For N = k + 1 there are the root d ∈ (1/2, 2/3) of d³ + 3Nd = 2N, the zero
z₀ = N + d of q_{k,3}, the quadratic factor z² − 2uz + v of the complex pair (u² < v = |z_±|², z₀² < v),
the residue R₀ = −1/(z₀ q'(z₀)) > 0 and K = R₀² · 8N(N+d)²(1−d)²/(d³[(N+d)² + 3N(1−d)][3N+d²+d+1])
(NOTE.md §2's (2|R₊|)²) with 0 ≤ K < R₀², such that (c_n − R₀ z₀^{−n})² ≤ K z₀^{−2n} for all n;
hence c_n ≥ (R₀ − r) z₀^{−n} for every r ≥ 0 with K ≤ r² (r = √K = 2|R₊| < R₀). -/
theorem c_quant (k : ℕ) (N : ℝ) (hN : N = k + 1) :
    ∃ d z₀ u v R₀ K : ℝ,
      (1 / 2 < d ∧ d < 2 / 3 ∧ d ^ 3 + 3 * N * d = 2 * N) ∧
      z₀ = N + d ∧ qR k z₀ = 0 ∧
      (∀ z, N * (N + 1) * (N + 2) * qR k z = (z₀ - z) * (z ^ 2 - 2 * u * z + v)) ∧
      u ^ 2 < v ∧ z₀ ^ 2 < v ∧
      R₀ = -1 / (z₀ * qR' k z₀) ∧ 0 < R₀ ∧
      K = R₀ ^ 2 * (8 * N * (N + d) ^ 2 * (1 - d) ^ 2
        / (d ^ 3 * ((N + d) ^ 2 + 3 * N * (1 - d)) * (3 * N + d ^ 2 + d + 1))) ∧
      0 ≤ K ∧ K < R₀ ^ 2 ∧
      (∀ n, ((c k n : ℝ) - R₀ * (1 / z₀) ^ n) ^ 2 ≤ K * ((1 / z₀) ^ n) ^ 2) ∧
      (∀ n r, 0 ≤ r → K ≤ r ^ 2 → (R₀ - r) * (1 / z₀) ^ n ≤ (c k n : ℝ)) := by
