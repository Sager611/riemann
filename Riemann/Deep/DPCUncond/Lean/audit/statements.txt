-- ===== Gaps.lean

/-- Entries of the sorted list x 0 ≤ … ≤ x (n − 1) that share their value with another entry. -/
noncomputable def shared (x : ℕ → ℝ) (n : ℕ) : Finset ℕ :=
  (range n).filter (fun i => ∃ j ∈ range n, j ≠ i ∧ x j = x i)

/-- Zero gaps: indices i with i + 1 < n and x (i + 1) = x i. -/
noncomputable def zeroGaps (x : ℕ → ℝ) (n : ℕ) : Finset ℕ :=
  (range n).filter (fun i => i + 1 < n ∧ x (i + 1) = x i)

/-- **Step 2, list form**: #(entries sharing their value with another entry) ≤ 2·#(zero gaps). -/
theorem card_shared_le (x : ℕ → ℝ) (n : ℕ) (hx : ∀ i, i + 1 < n → x i ≤ x (i + 1)) :
    (shared x n).card ≤ 2 * (zeroGaps x n).card

/-- Real parts β n of the zeros, in pplean's ordinate order (Zeros.γ, with multiplicity). The functional
equation ρ ↦ 1 − ρ̄ permutes the zeros with multiplicity: an off-line zero (β ≠ 1/2) has a partner index with the
same ordinate and real part 1 − β. -/
def Reflect (Z : Zeros) (β : ℕ → ℝ) : Prop :=
  ∀ n : ℕ, 1 ≤ n → β n ≠ 1 / 2 → ∃ m : ℕ, 1 ≤ m ∧ Z.γ m = Z.γ n ∧ β m = 1 - β n

/-- The n-th zero is off-line or multiple. -/
def Exc (Z : Zeros) (β : ℕ → ℝ) (n : ℕ) : Prop :=
  β n ≠ 1 / 2 ∨ ∃ m : ℕ, 1 ≤ m ∧ m ≠ n ∧ Z.γ m = Z.γ n ∧ β m = β n

/-- Off-line or multiple zeros with T ≤ γ ≤ 2T. -/
noncomputable def excSet (Z : Zeros) (β : ℕ → ℝ) (T : ℝ) : Finset ℕ :=
  (Icc 1 (Z.N (2 * T))).filter (fun n => T ≤ Z.γ n ∧ Z.γ n ≤ 2 * T ∧ Exc Z β n)

/-- Z_off ∩ {T ≤ γ ≤ 2T} (zeros with β > 1/2): PROOF Step 6a's classes S_a are subsets. -/
noncomputable def offSet (Z : Zeros) (β : ℕ → ℝ) (T : ℝ) : Finset ℕ :=
  (Icc 1 (Z.N (2 * T))).filter (fun n => T ≤ Z.γ n ∧ Z.γ n ≤ 2 * T ∧ 1 / 2 < β n)

/-- 𝒩(T) of Theorem U: n with T ≤ γ_n ≤ γ_{n+1} ≤ 2T and γ_{n+1} − γ_n < 2πφ/log T, unfolded with the zero
model's sc T = log T/(2π): (γ_{n+1} − γ_n)·sc T < φ. (γ_n ≤ γ_{n+1} is Zeros.Ordered.mono; the ambient
Icc 1 N(2T) cuts nothing, since γ_{n+1} ≤ 2T forces n + 1 ≤ N(2T).) -/
noncomputable def gapSet (Z : Zeros) (φ T : ℝ) : Finset ℕ :=
  (Icc 1 (Z.N (2 * T))).filter
    (fun n => T ≤ Z.γ n ∧ Z.γ (n + 1) ≤ 2 * T ∧ (Z.γ (n + 1) - Z.γ n) * Z.sc T < φ)

/-- **Step 2 (zero model)**: #(off-line or multiple zeros with T ≤ γ ≤ 2T) ≤ 2𝒩(T). -/
theorem card_exc_le {Z : Zeros} (hZ : Z.Ordered) {β : ℕ → ℝ} (hβ : Reflect Z β) {φ : ℝ} (hφ : 0 < φ)
    (T : ℝ) : (excSet Z β T).card ≤ 2 * (gapSet Z φ T).card

/-- Step 2 as used in 6a: #(Z_off ∩ {T ≤ γ ≤ 2T}) ≤ 2𝒩(T). -/
theorem card_off_le {Z : Zeros} (hZ : Z.Ordered) {β : ℕ → ℝ} (hβ : Reflect Z β) {φ : ℝ} (hφ : 0 < φ)
    (T : ℝ) : (offSet Z β T).card ≤ 2 * (gapSet Z φ T).card

/-- E = (1 − 1/w)(1/2 − 1/(3r)) (PROOF 6e). -/
noncomputable def Eexp (w r : ℝ) : ℝ := (1 - 1 / w) * (1 / 2 - 1 / (3 * r))

/-- 6e: the σ-exponents of A₁, A₂, A₃ sum to 3E. -/
theorem exp_mean (w r : ℝ) :
    (1 - 1 / w) / 4 + (1 - 1 / w) / 4 + (1 - 1 / w) * (1 - 1 / r) = Eexp w r + Eexp w r + Eexp w r

/-- 6e, Hölder with exponents (3, 3, 3), as exponent bookkeeping on abstract nonnegative reals. With s e = σ^e
(any s with s(x + y) = s x · s y, s ≥ 0), B = (a + 2)^C, g = e^{θ(a+1)}:
A₁ ≤ K T B σ^{(1−1/w)/4}, A₂ ≤ K T B g³ σ^{(1−1/w)/4}, A₃ ≤ K T B σ^{(1−1/w)(1−1/r)} and 𝓗³ ≤ A₁A₂A₃
give 𝓗 ≤ K T B g σ^E. -/
theorem holder_bookkeeping {s : ℝ → ℝ} (hs : ∀ x y, s (x + y) = s x * s y) (hs0 : ∀ x, 0 ≤ s x)
    {w r K T B g A₁ A₂ A₃ H : ℝ} (hK : 0 ≤ K) (hT : 0 ≤ T) (hB : 0 ≤ B) (hg : 0 ≤ g)
    (hA₂ : 0 ≤ A₂) (hA₃ : 0 ≤ A₃)
    (h1 : A₁ ≤ K * T * B * s ((1 - 1 / w) / 4))
    (h2 : A₂ ≤ K * T * B * g ^ 3 * s ((1 - 1 / w) / 4))
    (h3 : A₃ ≤ K * T * B * s ((1 - 1 / w) * (1 - 1 / r)))
    (hH : H ^ 3 ≤ A₁ * A₂ * A₃) :
    H ≤ K * T * B * g * s (Eexp w r)

/-- 6e–6f: for every θ < 1/2, δ = x, w = 1/x, r = 1/(3x) with x = (1/2 − max θ 0)/4 give 0 < δ < 1, w, r > 1 and
θ < (1 − δ)E, i.e. κE > θ for Jutila's slope κ = 1 − δ. -/
theorem exists_params {θ : ℝ} (hθ : θ < 1 / 2) :
    ∃ δ w r : ℝ, 0 < δ ∧ δ < 1 ∧ 1 < w ∧ 1 < r ∧ θ < (1 - δ) * Eexp w r

/-- Abstract exp and log: the four facts of Mathlib's Real.exp, Real.log used here. -/
structure ExpLog (ex lg : ℝ → ℝ) : Prop where
  add : ∀ x y : ℝ, ex (x + y) = ex x * ex y

  mono : StrictMono ex
  ex_lg : ∀ x : ℝ, 0 < x → ex (lg x) = x

lemma ExpLog.zero (h : ExpLog ex lg) : ex 0 = 1

/-- **The tail of 6f**: for q ≥ 0 and 0 ≤ ρ < 1, Σ_{A<a≤M} (a + 2)^C q ρ^a ≤ τ for all A ≥ N, uniformly in M. -/
theorem tail_small (C : ℕ) {q ρ : ℝ} (hq : 0 ≤ q) (hρ0 : 0 ≤ ρ) (hρ1 : ρ < 1) :
    ∀ τ : ℝ, 0 < τ → ∃ N : ℕ, ∀ A M : ℕ, N ≤ A →
      ∑ a ∈ Ioc A M, ((a : ℝ) + 2) ^ C * q * ρ ^ a ≤ τ

-- ===== Phi.lean

/-- A(ε) = ⌊c log(1/ε)⌋ (PROOF.md Repair 6). -/
noncomputable def Aeps (lg : ℝ → ℝ) (c ε : ℝ) : ℕ := ⌊c * lg (1 / ε)⌋₊

/-- Φ(ε) of PROOF 6f, a-sum truncated at M. -/
noncomputable def Phi (ex lg : ℝ → ℝ) (C : ℕ) (θ κ E c ε : ℝ) (M : ℕ) : ℝ :=
  ((Aeps lg c ε : ℝ) + 2) ^ C * ex θ ^ (Aeps lg c ε + 1) * ex (lg (3 * ε) * E)
    + ∑ a ∈ Ioc (Aeps lg c ε) M, ((a : ℝ) + 2) ^ C * ex θ * ex (θ - κ * E) ^ a

/-- **Φ(ε) → 0 as ε → 0** (PROOF 6f, Repair 6), uniformly in the truncation M. -/
theorem Phi_small (h : ExpLog ex lg) (C : ℕ) {θ κ E c : ℝ} (hE : 0 ≤ E) (hκ : θ < κ * E) (hc : 0 < c)
    (hcθ : c * θ < E) : ∀ τ : ℝ, 0 < τ → ∃ ε₁ : ℝ, 0 < ε₁ ∧ ∀ ε : ℝ, 0 < ε → ε ≤ ε₁ → ∀ M : ℕ,
      Phi ex lg C θ κ E c ε M ≤ τ

  I : ℝ → ℝ

  /-- ∫_T^{2T} |P(V_φ)|⁴ (Step 5c) -/

/-- Step 6 (6a–6f; density N(σ, T) ≪ T^{1−κ(σ−1/2)} log T with Jutila's κ = 1 − δ, Lz l.223–225): for all
admissible δ, w, r (θ < (1 − δ)E) there are C, K, independent of ε, such that for every c ∈ (0, E/θ): if

def Step6 (Z : Zeros) (β : ℕ → ℝ) (ex lg : ℝ → ℝ) (θ : ℝ) (D : Data) : Prop :=

/-- Step 1 (Im V counts all zeros, so Im V ∈ {0, 1} outside E_T, where Re P(V) ≥ 0 by the two-line signs

/-- Step 7, Hölder with exponents (2, 4, 4), G_T bounded (DPC l.776–780):

/-- Step 7, covering (DPC l.774, 782): meas E_T ≤ h𝒩(T) + h with h = 2πφ/log T = φ/sc T (a consecutive pair with

/-- The zero-density theorem, uniform for σ ≥ 1/2 (Jutila 1983, slope κ = 1 − δ for every δ > 0; Conrey 8/7 − δ;
as quoted Lz 2611.14486 l.221–230): N(σ, 2T) ≪_δ T^{1−κ(σ−1/2)} log T at σ = 1/2 + a/log T gives
#{T ≤ γ ≤ 2T, β ≥ 1/2 + a/log T} ≤ C_d T log T e^{−κa}. -/
def Density (Z : Zeros) (β : ℕ → ℝ) (ex lg : ℝ → ℝ) : Prop :=
  ∀ δ : ℝ, 0 < δ → δ < 1 → ∃ Cd T₀ : ℝ, 0 < Cd ∧ ∀ T : ℝ, T₀ ≤ T → ∀ a : ℕ,
    ((deepSet Z β lg T a).card : ℝ) ≤ Cd * T * lg T * ex (-((1 - δ) * a))

/-- Steps 6a–6e at the level of the depth classes S_a: for all w, r > 1 there are C, K and class totals
Ha T a = ∫|R_T||R*|(1 + |V_L| + 2Ñ)^D dμ_a and A₁ = ∫|R_T|³dμ_a, A₂ = ∫|R*|³dμ_a, A₃ = ∫(1 + |V_L| + 2Ñ)^{3D}dμ_a
such that, for large T, (6a) |𝓗_T| ≤ Σ_{a≤M(T)} Ha T a + o(T) (each merged cut belongs to the class of its deepest
zero; zeros outside [T, 2T] give o(1), Step 3), (6e) Ha³ ≤ A₁A₂A₃ (Hölder) and, whenever |S_a| ≤ s T log T,
A₁ ≤ K T (a+2)^C s^{(1−1/w)/4}, A₂ ≤ K T (a+2)^C e^{3θ(a+1)} s^{(1−1/w)/4}, A₃ ≤ K T (a+2)^C s^{(1−1/w)(1−1/r)}
(6b–6d: Poisson representations, sparse kernels, Fujii's moments, Lz Lemma DoubleSumZeros l.251–283). -/
def Step6Classes (Z : Zeros) (β : ℕ → ℝ) (ex lg : ℝ → ℝ) (θ : ℝ) (D : Data) : Prop :=
  ∀ w r : ℝ, 1 < w → 1 < r → ∃ (C : ℕ) (K : ℝ), 0 ≤ K ∧
    ∃ (Ha A₁ A₂ A₃ : ℝ → ℕ → ℝ) (Mt : ℝ → ℕ), ∀ η : ℝ, 0 < η → ∃ T₀ : ℝ, ∀ T : ℝ, T₀ ≤ T →
      |D.Hp T| ≤ ∑ a ∈ range (Mt T + 1), Ha T a + η * T ∧
      ∀ a : ℕ, 0 ≤ A₂ T a ∧ 0 ≤ A₃ T a ∧ Ha T a ^ 3 ≤ A₁ T a * A₂ T a * A₃ T a ∧
        ∀ s : ℝ, 0 < s → ((classSet Z β lg T a).card : ℝ) ≤ s * T * lg T →
          A₁ T a ≤ K * T * ((a : ℝ) + 2) ^ C * ex (lg s * ((1 - 1 / w) / 4)) ∧
          A₂ T a ≤ K * T * ((a : ℝ) + 2) ^ C * (ex θ ^ (a + 1)) ^ 3 * ex (lg s * ((1 - 1 / w) / 4)) ∧
          A₃ T a ≤ K * T * ((a : ℝ) + 2) ^ C * ex (lg s * ((1 - 1 / w) * (1 - 1 / r)))

/-- **Step 6 from the density theorem and the class bounds** (6a counts, 6e bookkeeping and 6f sum proved). -/
theorem step6_of_classes (hEL : ExpLog ex lg) {Z : Zeros} {β : ℕ → ℝ} {θ : ℝ} (hθ : 0 ≤ θ) {D : Data}
    (hden : Density Z β ex lg) (h6c : Step6Classes Z β ex lg θ D) : Step6 Z β ex lg θ D

/-- Step 5a, Montgomery–Vaughan's mean value theorem for R_T² = Σ b_n n^{−it} (length T^{2θ}) with Σ|b_n|² ≤ B
(Mertens, the T^{η/2}-rough support): ∫|R_T|⁴ ≤ (2T + K T^{2θ}) B, T^{2θ} = e^{2θ log T}. -/
def MV5a (ex lg : ℝ → ℝ) (θ : ℝ) (D : Data) : Prop :=
  ∃ Kmv B T₀ : ℝ, ∀ T : ℝ, T₀ ≤ T → 0 ≤ D.R4 T ∧ D.R4 T ≤ (2 * T + Kmv * ex (lg T * (2 * θ))) * B

/-- Step 5c: ∫|P(V)|⁴ ≪ T (M. Riesz DPC l.641, Fujii Lz l.242–249, density; 5b–5c). -/
def Step5c (D : Data) : Prop :=
  ∃ K₂ T₀ : ℝ, ∀ T : ℝ, T₀ ≤ T → 0 ≤ D.P4 T ∧ D.P4 T ≤ K₂ * T

/-- **Step 5 from Montgomery–Vaughan**: T^{2θ} ≤ T for θ ≤ 1/2, so ∫|R_T|⁴ ≪ T. -/
theorem step5_of (hEL : ExpLog ex lg) {θ : ℝ} (hθ : θ ≤ 1 / 2) {D : Data} (h5a : MV5a ex lg θ D)
    (h5c : Step5c D) : Step5 D

/-- The finer named inputs for one Step-0 construction. -/
structure InputsFine (Z : Zeros) (β : ℕ → ℝ) (ex lg : ℝ → ℝ) (φ θ : ℝ) (D : Data) : Prop where
  step3 : Step3 D
  main : MainTerm D
  step6c : Step6Classes Z β ex lg θ D
  step1 : Step1Pos D
  holder : Holder7 D
  mv5a : MV5a ex lg θ D
  step5c : Step5c D
  cover : Covering Z φ D

/-- DPCInputs with Step 6 replaced by the class-level inputs and Step 5a by Montgomery–Vaughan. -/
def DPCInputsFine (Z : Zeros) (β : ℕ → ℝ) (ex lg : ℝ → ℝ) : Prop :=
  ∀ φ θ : ℝ, 0 < φ → φ < 1 → 0 < θ → θ < 1 / 2 → 0 < φ * θ → φ * θ < 1 → 1 < φ * (1 + θ) →
    φ * (1 + θ) < 2 → ∃ D : Data, InputsFine Z β ex lg φ θ D

theorem inputs_of_fine (hEL : ExpLog ex lg) {Z : Zeros} {β : ℕ → ℝ} (hden : Density Z β ex lg)
    (hin : DPCInputsFine Z β ex lg) : DPCInputs Z β ex lg
