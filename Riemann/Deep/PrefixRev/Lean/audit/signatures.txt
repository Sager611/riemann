@PrefixRev.phi_P : ∀ {n m k θ : ℕ},
  n = m + k + 1 →
    k < m →
      (k + 1) * θ % (n + 1) = 1 →
        ∀ (x : Fin n),
          PrefixRev.phi n θ ((PrefixRev.P n m k) x) = if PrefixRev.phi n θ x < n then PrefixRev.phi n θ x + 1 else 1

    k < m →
      (k + 1) * θ % (n + 1) = 1 → ∀ (x : Fin n), PrefixRev.phi n θ ((PrefixRev.rev n n) x) = n + 1 - PrefixRev.phi n θ x

    k < m →
      (k + 1) * θ % (n + 1) = 1 →
        ∀ (x : Fin n),

@PrefixRev.G_le_cls : ∀ {n m k : ℕ},
  n = m + k + 1 → k < m → PrefixRev.G n m k ≤ PrefixRev.fibreFix fun x => PrefixRev.cls ((k + 1).gcd (m + 1)) x
def PrefixRev.fibreFix : {n : ℕ} → {β : Type} → (Fin n → β) → Subgroup (Equiv.Perm (Fin n)) :=
fun {n} {β} b => { carrier := {g | ∀ (x : Fin n), b (g x) = b x}, mul_mem' := ⋯, one_mem' := ⋯, inv_mem' := ⋯ }
def PrefixRev.fibreStab : {n : ℕ} → {β : Type} → (Fin n → β) → Subgroup (Equiv.Perm (Fin n)) :=
fun {n} {β} b =>
  { carrier := {g | ∀ (x y : Fin n), b x = b y ↔ b (g x) = b (g y)}, mul_mem' := ⋯, one_mem' := ⋯, inv_mem' := ⋯ }
def PrefixRev.phi : (n : ℕ) → ℕ → Fin n → ℕ :=
fun n θ x => θ * (↑x + 1) % (n + 1)
def PrefixRev.P : (n : ℕ) → ℕ → ℕ → Equiv.Perm (Fin n) :=
fun n m k => PrefixRev.rev n k * PrefixRev.rev n n * PrefixRev.rev n m
