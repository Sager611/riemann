def ORf (L : ℝ → ℝ) (E : ℝ) (n : ℕ) : Prop := (σ 1 n : ℝ) / n < E / 2 * L (L n)
def WYIneqF (L : ℝ → ℝ) (E : ℝ) (n : ℕ) : Prop := (σ 1 n : ℝ) / (n * L (L n)) < E / 2
def WYTheoremFor (RH : Prop) (L : ℝ → ℝ) (E : ℝ) : Prop := RH ↔ ∀ n : ℕ, Odd n → c0 ≤ n → WYIneqF L E n
structure LogLike (L : ℝ → ℝ) : Prop where
  map_mul : ∀ x y : ℝ, 0 < x → 0 < y → L (x * y) = L x + L y          -- Real.log_mul
  mono : ∀ x y : ℝ, 0 < x → x ≤ y → L x ≤ L y                         -- Real.log_le_log
  one_sub_inv_le : ∀ x : ℝ, 0 < x → 1 - x⁻¹ ≤ L x                    -- Real.one_sub_inv_le_log_of_pos
  series_lo : ∀ a : ℝ, 0 ≤ a → ∀ m : ℕ, lsum a m ≤ L (1 + a)          -- from Real.hasSum_log_one_add
  series_hi : ∀ a : ℝ, 0 ≤ a → ∀ m : ℕ, L (1 + a) ≤ lsum a m + ltail a m
  -- lsum a m = Σ_{k<m} 2/(2k+1) y^(2k+1), ltail a m = 2 y^(2m+1)/((2m+1)(1−y²)), y = a/(a+2)
